    <canvas id="c"></canvas>

    <audio id="crackle" src="https://upload.wikimedia.org/wikipedia/commons/4/45/Bonfire.ogg" loop preload="auto"></audio>

    <div class="hud">

      <div style="display:flex; gap:12px; align-items:center;">

        <div class="badge"><span style="opacity:.8">Fuel: </span><strong id="fuel">0</strong></div>

        <div class="badge"><span style="opacity:.8">Wind: </span><strong id="wind">0</strong></div>

        <div class="badge"><span style="opacity:.8">Heat: </span><strong id="heat">0</strong></div>

      </div>

      <div style="opacity:.9" id="clock">0:00</div>

    </div>

    <div class="footer">

      <div class="hint">Tap to add wood · Swipe ←/→ wind · Swipe ↑ puff</div>

      <div style="margin-top:8px"><button class="btn" id="resetBtn">Reset</button></div>

    </div>

  <script>

  (function(){

    const canvas = document.getElementById('c');

    const g = canvas.getContext('2d');

    const audio = document.getElementById('crackle');

    const hudFuel = document.getElementById('fuel');

    const hudWind = document.getElementById('wind');

    const hudHeat = document.getElementById('heat');

    const hudClock = document.getElementById('clock');

    const resetBtn = document.getElementById('resetBtn');


    function clamp(v, lo, hi){ return Math.max(lo, Math.min(hi, v)); }

    function formatTime(s){ const m = Math.floor(s/60); const ss = s % 60; return m + ':' + String(ss).padStart(2,'0'); }


    class Game {

      constructor(ctx, width, height, dpr){

        this.ctx = ctx; this.W = width; this.H = height; this.dpr = dpr;

        this.time = 0; this.wind = 0; this.fuel = 40; this.heat = 0;

        this.pool = []; this.particles = []; this.maxParticles = 450;

        this.spawnCooldown = 0; this.holdToFeed = 0;

      }

      addFuel(a=4){ this.fuel = Math.min(100, this.fuel + a); }

      puffAir(){ for(let i=0;i<60;i++) this.spawnFlame(true); this.heat += 6; }

      spawnFlame(puff=false){

        if(this.particles.length >= this.maxParticles) return;

        const p = this.pool.pop() || {};

        const cx = this.W/2; const baseY = this.H*0.82; const spread = Math.max(18, 60 + this.wind*24);

        p.x = cx + (Math.random()-0.5)*spread; p.y = baseY + (Math.random()-0.5)*6;

        const up = puff ? 2.6 : 2.0 + Math.random()*0.9;

        p.vx = this.wind*(0.3 + Math.random()*0.5) + (Math.random()-0.5)*0.2;

        p.vy = -up - Math.random()*0.6;

        p.life = 0; p.max = 28 + Math.random()*22 + (puff?12:0);

        p.size = 12 + Math.random()*14 + (puff?6:0);

        p.h = 15 + Math.random()*25; p.s = 85 + Math.random()*10; p.l = 50 + Math.random()*10;

        p.alpha = 0.85; p.type = 'flame';

        this.particles.push(p);

      }

      spawnEmber(){

        if(this.particles.length >= this.maxParticles) return;

        const e = this.pool.pop() || {};

        const cx = this.W/2; const baseY = this.H*0.82;

        e.x = cx + (Math.random()-0.5)*30; e.y = baseY - 10 + (Math.random()-0.5)*10;

        e.vx = this.wind*(0.5 + Math.random()*0.6) + (Math.random()-0.5)*0.2;

        e.vy = -1.6 - Math.random()*0.8;

        e.life = 0; e.max = 120 + Math.random()*100; e.size = 2 + Math.random()*2; e.alpha=1; e.type='ember';

        this.particles.push(e);

      }

      update(dt){

        this.time += dt;

        const targetHeat = Math.max(0, Math.min(100, this.fuel*1.2 + Math.abs(this.wind)*8));

        this.heat += (targetHeat - this.heat) * 0.04;

        const burnRate = 0.005 + this.heat * 0.0004;

        this.fuel = Math.max(0, this.fuel - burnRate*dt);

        this.spawnCooldown -= dt;

        const spawnInterval = 8 + (100 - this.heat) * 0.2;

        while(this.spawnCooldown <= 0 && this.heat > 3){

          this.spawnFlame(false);

          if(Math.random() < 0.25) this.spawnEmber();

          this.spawnCooldown += spawnInterval;

        }

        for(let i=this.particles.length-1;i>=0;i--){

          const p = this.particles[i];

          p.life += dt*0.06;

          if(p.type==='flame'){

            p.vx += this.wind * 0.002 * dt; p.vy -= 0.0006 * dt;

            p.x += p.vx * (dt*0.06); p.y += p.vy * (dt*0.06);

            p.alpha = Math.max(0, 0.9 - p.life/p.max); p.size *= 0.997;

            if(p.life>p.max || p.alpha<=0 || p.y < this.H*0.1) this.recycle(i);

          } else {

            p.vx += this.wind * 0.001 * dt; p.vy -= 0.0002 * dt;

            p.x += p.vx * (dt*0.06); p.y += p.vy * (dt*0.06);

            p.alpha = Math.max(0, 1 - p.life/p.max);

            if(p.life>p.max || p.alpha<=0 || p.y < -10) this.recycle(i);

          }

        }

        if(this.holdToFeed>0){ this.holdToFeed -= dt; if(this.holdToFeed<0) this.holdToFeed=0; this.addFuel(0.02 * dt * 0.06); }

        return { fuel:this.fuel, wind:this.wind, time:this.time, heat:this.heat };

      }

      recycle(i){ const p=this.particles[i]; this.pool.push(p); this.particles[i]=this.particles[this.particles.length-1]; this.particles.pop(); }

      draw(){

        const g=this.ctx, W=this.W, H=this.H;

        const sky=g.createLinearGradient(0,0,0,H); sky.addColorStop(0,'#05060a'); sky.addColorStop(1,'#0b0e14'); g.fillStyle=sky; g.fillRect(0,0,W,H);

        g.fillStyle='#151512'; g.fillRect(0,H*0.8,W,H*0.2);

        g.save(); g.translate(W/2, H*0.83); g.rotate(-0.15); roundRect(g,-90,-10,180,20,10); g.fillStyle='#3a2f21'; g.fill(); g.restore();

        g.save(); g.translate(W/2, H*0.83); g.rotate(0.25); roundRect(g,-90,-10,180,20,10); g.fillStyle='#3b2a1f'; g.fill(); g.restore();

        g.globalCompositeOperation='lighter';

        for(let i=0;i<this.particles.length;i++){

          const p=this.particles[i];

          if(p.type==='flame'){

            const grad=g.createRadialGradient(p.x,p.y,0,p.x,p.y,p.size);

            const c1=`hsla(${p.h}, ${p.s}%, ${p.l}%, ${p.alpha})`;

            const c2=`hsla(${Math.max(5,p.h-10)}, ${p.s}%, ${Math.max(30,p.l-10)}%, 0)`;

            grad.addColorStop(0,c1); grad.addColorStop(1,c2);

            g.fillStyle=grad; g.beginPath(); g.arc(p.x,p.y,p.size,0,Math.PI*2); g.fill();

          } else {

            g.globalAlpha = p.alpha*0.8; g.beginPath(); g.arc(p.x,p.y,p.size,0,Math.PI*2); g.fillStyle='#ffb347'; g.fill(); g.globalAlpha = 1;

          }

        }

        g.globalCompositeOperation='source-over';

        const glow=g.createRadialGradient(W/2,H*0.83,0,W/2,H*0.83,120); glow.addColorStop(0,'rgba(255,180,80,0.06)'); glow.addColorStop(1,'rgba(255,180,80,0)'); g.fillStyle=glow; g.beginPath(); g.arc(W/2,H*0.83,120,0,Math.PI*2); g.fill();

      }

    }


    function roundRect(g,x,y,w,h,r){

      g.beginPath(); g.moveTo(x+r,y); g.arcTo(x+w,y,x+w,y+h,r); g.arcTo(x+w,y+h,x,y+h,r); g.arcTo(x,y+h,x,y,r); g.arcTo(x,y,x+w,y,r); g.closePath();

    }


    function fit(){

      const dpr=Math.max(1, Math.min(3, window.devicePixelRatio||1));

      const rect = canvas.getBoundingClientRect();

      canvas.width = Math.floor(rect.width*dpr); canvas.height = Math.floor(rect.height*dpr);

      g.setTransform(dpr,0,0,dpr,0,0);

      if(game){ game.W = rect.width; game.H = rect.height; game.dpr = dpr; }

    }


    const rootRect = document.getElementById('root').getBoundingClientRect();

    const dpr0 = Math.max(1, Math.min(3, window.devicePixelRatio||1));

    canvas.width = Math.floor(rootRect.width*dpr0); canvas.height = Math.floor(rootRect.height*dpr0);

    g.setTransform(dpr0,0,0,dpr0,0,0);


    let game = new Game(g, rootRect.width, rootRect.height, dpr0);

    let last = performance.now(); let raf = 0;

    function loop(t){

      const dt = Math.min(40, t - last); last = t;

      const state = game.update(dt); game.draw();

      hudFuel.textContent = Math.round(state.fuel);

      hudWind.textContent = Math.round(state.wind*100)/100;

      hudHeat.textContent = Math.round(state.heat);

      hudClock.textContent = formatTime(Math.floor(state.time/1000));

      raf = requestAnimationFrame(loop);

    }

    raf = requestAnimationFrame(loop);


    window.addEventListener('resize', fit, { passive:true });


    const gesture = { active:false, lastX:0, lastY:0, t0:0 };

    function unlockAudio(){ if(audio && audio.paused){ try{ audio.currentTime = 0; audio.play(); }catch(e){} } }


    canvas.addEventListener('touchstart', function(e){

      const t = e.touches[0]; gesture.active=true; gesture.lastX=t.clientX; gesture.lastY=t.clientY; gesture.t0=performance.now();

      game.holdToFeed = 9999; game.addFuel(2); unlockAudio();

    }, { passive:false });


    canvas.addEventListener('touchmove', function(e){

      e.preventDefault();

      const t = e.touches[0]; const dx = t.clientX - gesture.lastX; const dy = t.clientY - gesture.lastY;

      gesture.lastX = t.clientX; gesture.lastY = t.clientY;

      game.wind = clamp(game.wind + dx * 0.002, -1.2, 1.2);

      if(dy < -6) game.puffAir();

    }, { passive:false });


    canvas.addEventListener('touchend', function(){

      gesture.active=false; game.holdToFeed = 0;

      const dt = performance.now() - gesture.t0; if(dt < 200) game.addFuel(6);

    }, { passive:false });


    resetBtn.addEventListener('click', function(){

      game.fuel=40; game.wind=0; game.heat=0; game.time=0; game.particles.length=0; game.pool.length=0;

    });


    requestAnimationFrame(fit);

  })();

  </script>
